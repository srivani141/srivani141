import pandas as pd
import matplotlib.pyplot as plt
import os

# File to store expenses
file_name = "expenses.csv"

# Function to add an expense
def add_expense(date, category, amount, description):
    data = {
        "Date": [date],
        "Category": [category],
        "Amount": [amount],
        "Description": [description]
    }
    df = pd.DataFrame(data)
    
    if os.path.isfile(file_name):
        df.to_csv(file_name, mode='a', header=False, index=False)
    else:
        df.to_csv(file_name, index=False)
    
    print("✅ Expense Added Successfully!")

# Function to display all expenses
def show_expenses():
    if os.path.isfile(file_name):
        df = pd.read_csv(file_name)
        print("\n📋 All Expenses:")
        print(df)
    else:
        print("⚠️ No expenses found!")

# Function to show total expenses by category
def show_expenses_by_category():
    if os.path.isfile(file_name):
        df = pd.read_csv(file_name)
        category_total = df.groupby("Category")["Amount"].sum()
        print("\n📊 Total Expenses by Category:")
        print(category_total)

        # Plot pie chart
        category_total.plot(kind='pie', autopct='%1.1f%%', figsize=(6, 6))
        plt.title("Expenses by Category")
        plt.ylabel('')  # Remove default ylabel
        plt.show()
    else:
        print("⚠️ No expenses found!")

# Main menu
def main_menu():
    while True:
        print("\n📊 Personal Expense Tracker 📊")
        print("1. Add Expense")
        print("2. Show All Expenses")
        print("3. Show Expenses by Category")
        print("4. Exit")

        try:
            choice = int(input("Enter your choice (1-4): "))
        except ValueError:
            print("❌ Invalid input, please enter a number.")
            continue

        if choice == 1:
            date = input("Enter Date (YYYY-MM-DD): ")
            category = input("Enter Category (Food, Travel, Bills, Others): ")
            try:
                amount = float(input("Enter Amount: "))
            except ValueError:
                print("❌ Invalid amount, please enter a number.")
                continue
            description = input("Enter Description: ")
            add_expense(date, category, amount, description)

        elif choice == 2:
            show_expenses()

        elif choice == 3:
            show_expenses_by_category()

        elif choice == 4:
            print("👋 Exiting... Thank you for using Expense Tracker!")
            break

        else:
            print("❌ Invalid choice, please select between 1 and 4.")

# Run the application
if __name__ == "__main__":
    main_menu()
